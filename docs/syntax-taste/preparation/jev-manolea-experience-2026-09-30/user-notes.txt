fn bool valid_mime
    emits []
    given
        str name
        str mime
    asserts
        html: "index.html", "text/html" => ok true
        mismatch: "index.html", "text/javascript" => ok false
        script: "assets/app.js", "text/javascript" => ok true
    match call path::extension(name)
        ok str ext => match ext is ".html"
            false => match ext is ".css"
                false => match ext is ".js"
                    false => match ext is ".mjs"
                        false => match ext is ".json"
                            false => match ext is ".txt"
                                false => match ext is ".png"
                                    false => match ext is ".jpg"
                                        false => match ext is ".jpeg"
                                            false => match ext is ".webp"
                                                false => match ext is ".woff"
                                                    false => match ext is ".woff2"
                                                        false => ok false
                                                        true => ok mime is "font/woff2"
                                                    true => ok mime is "font/woff"
                                                true => ok mime is "image/webp"
                                            true => ok mime is "image/jpeg"
                                        true => ok mime is "image/jpeg"
                                    true => ok mime is "image/png"
                                true => ok mime is "text/plain"
                            true => ok mime is "application/json"
                        true => ok mime is "text/javascript"
                    true => ok mime is "text/javascript"
                true => ok mime is "text/css"
            true => ok mime is "text/html"
