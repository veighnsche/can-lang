import atexit, fcntl, json, os, pathlib, shutil, signal, time, uuid
EVIDENCE=pathlib.Path('/Users/vince/Projects/can-lang/docs/syntax-taste/preparation/native-can-test-preparatory-checks-2026-09-30')
RUN_ID='run-'+time.strftime('%Y%m%dT%H%M%S')+'-'+uuid.uuid4().hex[:6]
ROOT=pathlib.Path('/private/tmp/can-native-prep-'+RUN_ID)
INTENT=EVIDENCE/(RUN_ID+'-intent.json')
STATE={'kind':'can.preparatory.owner','run_id':RUN_ID,'root':str(ROOT),'owner_pid':os.getpid(),'owner_start':time.time(),'phase':'preparing','root_identity':None,'marker_identity':None,'safe_to_remove':True,'cleanup_errors':[]}
def save_state():
    INTENT.write_text(json.dumps(STATE,indent=2)+'\n')
def cleanup_storage():
    try:
        emergency=globals().get('emergency_cleanup')
        if emergency is not None: emergency()
        if not STATE['safe_to_remove']: raise RuntimeError('process cleanup unresolved; preserving owned root')
        if ROOT.exists():
            st=ROOT.lstat()
            if [st.st_dev,st.st_ino]!=STATE['root_identity']: raise RuntimeError('root identity changed')
            st=(ROOT/'.owner.json').lstat()
            if [st.st_dev,st.st_ino]!=STATE['marker_identity']: raise RuntimeError('marker identity changed')
            shutil.rmtree(ROOT)
        STATE['phase']='cleaned'
        STATE['root_removed']=not ROOT.exists()
    except Exception as exc:
        STATE['phase']='cleanup_failed'; STATE['cleanup_errors'].append(str(exc))
    save_state()
def interrupted(signum,frame): raise KeyboardInterrupt('handled interruption')
atexit.register(cleanup_storage)
signal.signal(signal.SIGTERM,interrupted)
signal.signal(signal.SIGINT,interrupted)
save_state()
ROOT.mkdir(mode=0o700)
st=ROOT.lstat(); STATE['root_identity']=[st.st_dev,st.st_ino]
LOCK=os.open(ROOT/'.owner.json',os.O_CREAT|os.O_EXCL|os.O_RDWR,0o600)
fcntl.flock(LOCK,fcntl.LOCK_EX|fcntl.LOCK_NB)
os.write(LOCK,json.dumps({'run_id':RUN_ID,'pid':os.getpid()}).encode())
st=os.fstat(LOCK); STATE['marker_identity']=[st.st_dev,st.st_ino]; save_state()
for name in ['owner.py','native-subject.ts','codec-subject.ts','descriptor-subject.ts','nonreader-subject.ts','browser-driver.ts']:
    shutil.copyfile(pathlib.Path('/private/tmp/can-native-prep-run-20260930T161737-8cab0b')/name, ROOT/name)
print(json.dumps({'preparation_root':str(ROOT),'run_id':RUN_ID,'owner_pid':os.getpid(),'state':'waiting_for_reviewed_instruments'}),flush=True)
try:
    command=input()
    if command!='run': raise RuntimeError('expected run command; no subjects launched')
    STATE['phase']='running';save_state()
    exec(compile((ROOT/'owner.py').read_text(),str(ROOT/'owner.py'),'exec'),globals())
except BaseException as exc:
    STATE['execution_exception']=type(exc).__name__+': '+str(exc); save_state()
    print(json.dumps({'execution_exception':STATE['execution_exception']}),flush=True)
